"""
================================================================================
INSTITUCIÓN FINANCIERA - DIRECCIÓN DE RIESGO DE CRÉDITO TDC
================================================================================
ARCHIVO:            <nombre_del_script>.py
DOMINIO:            [Originación / Líneas de Crédito / Cancelaciones / Monitoreo]
AUTOR:              [Nombre y Apellidos del Analista / Científico de Datos]
CORREO:             [nombre.apellido@bancoppel.com]
FECHA CREACIÓN:     2026-10-27
TICKET / JIRA:      POL-TDC-[1234]

OBJETIVO DE NEGOCIO:
    Evaluación de filtros duros de Buró y asignación de Límite de Crédito.

REGLAS CUANTITATIVAS / POLÍTICAS APLICADAS:
    1. Score Buró mínimo requerido: >= 655 puntos (BC Score).
    2. Nivel de endeudamiento (DTI) máximo: <= 40% del ingreso comprobable.
    3. MOP histórico en tarjeta de crédito: 00 (sin atrasos en los últimos 12 meses).

TABLAS Y FUENTES DE DATOS:
    - DWH_RIESGO.SOLICITUDES_DIGITALES_TDC
    - DWH_RIESGO.CONSULTAS_BURO_CREDITO

SALIDAS ESPERADAS:
    - Dictamen de crédito: 'APROBADO', 'RECHAZADO', 'REVISION_MANUAL'
    - Límite de Crédito Asignado ($ MXN)
================================================================================
"""

import sys
import logging
from typing import Dict, Any, Tuple

logging.basicConfig(level=logging.INFO, format="%(asctime)s [%(levelname)s] %(message)s")
logger = logging.getLogger("PoliticasTDC")

SCORE_MINIMO_ADMISION: int = 655
DTI_MAXIMO_PERMITIDO: float = 0.40

def evaluar_politica(solicitud: Dict[str, Any]) -> Tuple[str, str, float]:
    """Evalúa la solicitud y retorna dictamen, motivo y límite aprobado."""
    score = int(solicitud['bc_score'])
    ingreso = float(solicitud['ingreso_mensual'])
    pago_deuda = float(solicitud['pago_deuda_mensual'])

    if score < SCORE_MINIMO_ADMISION:
        return 'RECHAZADO', f'Score insuficiente ({score} < {SCORE_MINIMO_ADMISION})', 0.0

    dti = pago_deuda / ingreso if ingreso > 0 else 1.0
    if dti > DTI_MAXIMO_PERMITIDO:
        return 'RECHAZADO', f'Endeudamiento excesivo ({dti:.1%} > {DTI_MAXIMO_PERMITIDO:.1%})', 0.0

    limite = round(min((ingreso - pago_deuda) * 1.5, 50000.0), -2)
    return 'APROBADO', 'Cumple con scorecuts y DTI dentro de política', limite

if __name__ == "__main__":
    logger.info("=== Prueba sintética en memoria ===")
    ejemplo = {"bc_score": 690, "ingreso_mensual": 30000, "pago_deuda_mensual": 6000}
    dictamen, motivo, limite = evaluar_politica(ejemplo)
    logger.info(f"Resultado: {dictamen} | {motivo} | Línea: ${limite:,.2f}")
Componentes Obligatorios en SQL:
  • Cabecera de Auditoría: Archivo, autor, periodicidad, tablas fuente y tabla/vista destino.
  • Particionado Obligatorio: Filtrar siempre por FECHA_CORTE = CURRENT_DATE() para optimizar cómputo en Snowflake/Oracle.
  • Estructura con CTEs (WITH): Evita subconsultas anidadas y prohíbe el uso de SELECT *.

/*
================================================================================
INSTITUCIÓN FINANCIERA - DIRECCIÓN DE RIESGO DE CRÉDITO TDC
================================================================================
ARCHIVO:            <nombre_de_la_consulta>.sql
DOMINIO:            [Originación / Líneas de Crédito / Cancelaciones / Monitoreo]
AUTOR:              [Nombre y Apellidos del Analista / Científico de Datos]
CORREO:             [nombre.apellido@bancoppel.com]
FECHA CREACIÓN:     2026-10-27
MOTOR / DIALECTO:   Snowflake SQL / Oracle ANSI
PERIODICIDAD:       Mensual (Día hábil 3 de cada mes)
TICKET / JIRA:      POL-TDC-[1234]

OBJETIVO DE NEGOCIO:
    Extracción mensual de clientes elegibles para Campaña de Líneas de Crédito (CLI).

REGLAS DE FILTRADO Y ELEGIBILIDAD:
    1. Antigüedad de la cuenta >= 6 meses cumplidos.
    2. Sin morosidad en los últimos 6 meses (MOP = '00').
    3. Utilización promedio en los últimos 3 meses entre 50% y 85%.

TABLAS FUENTE (DATA WAREHOUSE):
    - DWH_RIESGO.MAESTRO_CUENTAS_TDC
    - DWH_RIESGO.FACTURACION_MENSUAL_TDC

TABLA / VISTA DESTINO:
    - BI_RIESGO.CAMPANA_CLI_CANDIDATOS_OCT2026
================================================================================
*/

WITH cuentas_elegibles AS (
    SELECT 
        numero_cuenta_hash,
        fecha_apertura,
        limite_credito_actual,
        DATEDIFF(month, fecha_apertura, CURRENT_DATE()) AS antiguedad_meses
    FROM DWH_RIESGO.MAESTRO_CUENTAS_TDC
    WHERE estatus_cuenta = 'ACTIVA'
      AND fecha_apertura <= DATEADD(month, -6, CURRENT_DATE())
),

metricas_comportamiento AS (
    SELECT 
        numero_cuenta_hash,
        AVG(saldo_promedio / NULLIF(limite_credito_promedio, 0)) AS utilizacion_promedio_3m,
        MAX(mop_cierre) AS max_mop_ultimos_6m
    FROM DWH_RIESGO.FACTURACION_MENSUAL_TDC
    WHERE fecha_corte >= DATEADD(month, -6, CURRENT_DATE())
    GROUP BY numero_cuenta_hash
)

SELECT 
    ce.numero_cuenta_hash,
    ce.antiguedad_meses,
    ce.limite_credito_actual,
    ROUND(mc.utilizacion_promedio_3m * 100, 2) AS pct_utilizacion_promedio,
    ROUND(ce.limite_credito_actual * 1.30, -2) AS nuevo_limite_propuesto,
    CURRENT_DATE()                             AS fecha_extraccion
FROM cuentas_elegibles ce
INNER JOIN metricas_comportamiento mc
    ON ce.numero_cuenta_hash = mc.numero_cuenta_hash
WHERE mc.max_mop_ultimos_6m = '00'
  AND mc.utilizacion_promedio_3m BETWEEN 0.50 AND 0.85;

Gobernanza Institucional

Los 4 Mandamientos de Convivencia y Seguridad

Reglas de cumplimiento estricto para garantizar la continuidad y auditoría de la Dirección de Riesgo:

1.

Prohibido subir Datos Personales (PII)

Por regulación de la CNBV y la Ley de Datos Personales, jamás agregues archivos .csv, .xlsx, .parquet ni credenciales en .env. Todo dato debe residir y consultarse dentro del Data Warehouse seguro.

2.

El Hábito Sagrado del "git pull" Diario

Todas las mañanas al abrir tu Visual Studio Code, el primer comando obligatorio antes de editar cualquier archivo siempre es git pull. Esto sincroniza en 2 segundos los cambios que tus compañeros subieron el día anterior.

3.

Independencia de Carpetas = Cero Conflictos

Trabaja únicamente en la carpeta de tu dominio. Si el analista de límites trabaja en lineas-credito/ y el de admisión en originacion/, Git fusiona ambos trabajos automáticamente sin pedir resolución de conflictos.

4.

Mensajes de Commit Claros y Semánticos

Evita mensajes ambiguos como "cambios" o "actualizacion". Usa la convención profesional: feat(lineas): ... para nuevas reglas, fix(originacion): ... para correcciones, y docs(readme): ... para documentación.
